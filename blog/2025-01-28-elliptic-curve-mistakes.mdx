# Elliptic Curve Cryptography: Common Implementation Pitfalls

> How to avoid the 12 most dangerous ECC mistakes that still break applications today.

## 1. Curve Selection

### Problem: Insecure Curves
- **Vulnerable**: secp112r1, secp128r1, secp160k1
- **Secure Choices**:
  - NIST P-256 (secp256r1)
  - Curve25519 (X25519 for ECDH)
  - secp256k1 (Bitcoin's curve)

### Code: Validating Curve Parameters
```python
def validate_curve_params(curve_name):
    SECURE_CURVES = {
        'secp256r1': {'field_size': 256, 'cofactor': 1},
        'X25519': {'field_size': 255, 'cofactor': 8},
        'secp256k1': {'field_size': 256, 'cofactor': 1}
    }
    
    if curve_name not in SECURE_CURVES:
        raise ValueError(f"Insecure curve: {curve_name}")
    
    return SECURE_CURVES[curve_name]
```

## 2. Point Validation

### The Missing Check
```python
# WRONG - Missing point validation
def unsafe_ecdh(private_key, peer_public_key):
    shared_secret = private_key * peer_public_key  # May be on small subgroup
    return shared_secret

# CORRECT - With validation
def safe_ecdh(private_key, peer_public_key, curve):
    if not is_point_on_curve(peer_public_key, curve):
        raise ValueError("Invalid public key")
    if not is_point_at_infinity(peer_public_key * curve.cofactor):
        raise ValueError("Invalid public key (small subgroup)")
    return private_key * peer_public_key
```

## 3. Side-Channel Attacks

### Vulnerable Implementation
```c
// Timing leak in scalar multiplication
void unsafe_scalar_mult(Point *result, const Scalar k, const Point *P) {
    Point Q = point_at_infinity();
    for (int i = 255; i >= 0; i--) {
        Q = point_double(Q);
        if ((k >> i) & 1) {
            Q = point_add(Q, P);  // Timing depends on bit value
        }
    }
    *result = Q;
}
```

### Fixed: Constant-Time Implementation
```c
// Montgomery Ladder for constant-time
void safe_scalar_mult(Point *result, const Scalar k, const Point *P) {
    Point R0 = point_at_infinity();
    Point R1 = *P;
    
    for (int i = 254; i >= 0; i--) {
        Point T = point_add(R0, R1);
        uint8_t bit = (k >> i) & 1;
        cswap(&R0, &R1, bit);
        R1 = point_add(R0, R1);
        R0 = point_double(R0);
        cswap(&R0, &R1, bit);
    }
    *result = R0;
}
```

## 4. Random Number Generation

### Common Mistakes
1. **Reusing Nonces** in ECDSA
   - **Vulnerable**: `r = (k*G).x mod n` (reused k)
   - **Secure**: `k = HMAC(privkey, message)` (RFC 6979)

2. **Biased Randomness**
   ```python
   # WRONG - Biased modulo reduction
   def gen_private_key():
       return random.getrandbits(256) % curve.n  # Bias in LSBs
   
   # CORRECT - Rejection sampling
   def gen_private_key_secure(curve):
       while True:
           k = random.getrandbits(256)
           if 0 < k < curve.n:
               return k
   ```

## 5. Signature Verification

### The Forgery Attack
```python
# WRONG - Verify without checking r,s range
def verify_broken(signature, message, public_key):
    r, s = signature
    if r == 0 or s == 0:  # Insufficient checks!
        return False
    w = modinv(s, curve.n)
    u1 = (hash(message) * w) % curve.n
    u2 = (r * w) % curve.n
    P = u1 * G + u2 * public_key
    return r == (P.x % curve.n)

# Attack: (r, s) = (n, n) verifies for any message!
```

### Secure Verification
```python
def verify_secure(signature, message, public_key, curve):
    r, s = signature
    # Check bounds
    if not (0 < r < curve.n and 0 < s < curve.n):
        return False
    # Verify point is on curve (if not already validated)
    if not is_point_on_curve(public_key, curve):
        return False
    
    w = modinv(s, curve.n)
    u1 = (hash(message) * w) % curve.n
    u2 = (r * w) % curve.n
    P = u1 * G + u2 * public_key
    
    if P == point_at_infinity():
        return False
        
    return r == (P.x % curve.n)
```

## 6. Curve25519 Pitfalls

### Clamping
```c
// WRONG - Missing key clamping
void unsafe_x25519_keygen(uint8_t private_key[32]) {
    randombytes(private_key, 32);
    // Missing: private_key[0] &= 248;
    // Missing: private_key[31] &= 127;
    // Missing: private_key[31] |= 64;
}

// CORRECT - Clamped key
typedef struct { uint8_t b[32]; } x25519_key;

void x25519_keygen(x25519_key *private_key) {
    randombytes(private_key->b, 32);
    private_key->b[0] &= 248;    // Clear bits 0,1,2
    private_key->b[31] &= 127;   // Clear bit 255
    private_key->b[31] |= 64;    // Set bit 254
}
```

## 8. Timing Leaks in Scalar Inversion

### Vulnerable Inversion
```python
def unsafe_inv(a, p):
    # Extended Euclidean Algorithm
    # Leaks timing based on number of iterations
    old_r, r = a, p
    old_s, s = 1, 0
    
    while r != 0:
        quotient = old_r // r
        old_r, r = r, old_r - quotient * r
        old_s, s = s, old_s - quotient * s
    
    return old_s % p
```

### Constant-Time Inversion
```python
def safe_inv(a, p):
    # Fermat's Little Theorem: a^(p-2) ≡ a^-1 mod p
    # Using square-and-multiply in constant time
    result = 1
    a = a % p
    exponent = p - 2
    
    while exponent > 0:
        # Always perform both operations
        square = (result * result) % p
        multiply = (result * a) % p
        
        # Conditional move based on exponent bit
        result = multiply if (exponent & 1) else square
        exponent >>= 1
    
    return result
```

## 9. Fault Injection Attacks

### The Bellcore Attack
```python
# Vulnerable RSA-CRT implementation
def sign_rsa_crt(message, p, q, d):
    dp = d % (p - 1)
    dq = d % (q - 1)
    qinv = modinv(q, p)
    
    # Compute m^d mod p and m^d mod q
    s1 = pow(message, dp, p)
    s2 = pow(message, dq, q)  # Fault injection here!
    
    # Recombine with CRT
    h = (qinv * (s1 - s2)) % p
    return s2 + h * q

# Attacker can recover private key from a single faulty signature
```

### Countermeasures
1. **Verify Before Output**
   ```python
   def safe_sign(message, p, q, d, n):
       signature = sign_rsa_crt(message, p, q, d)
       # Verify signature before returning
       if pow(signature, 65537, n) == message:
           return signature
       raise RuntimeError("Fault detected")
   ```

2. **Add Random Blinding**
   ```python
   def blind_sign(message, p, q, d, n):
       # Add random blinding factor
       r = random.getrandbits(32)
       blinded = (message * pow(r, 65537, n)) % n
       
       # Sign blinded message
       s = sign_rsa_crt(blinded, p, q, d)
       
       # Remove blinding
       return (s * modinv(r, n)) % n
   ```

## 10. Protocol-Level Vulnerabilities

### Invalid Curve Attack
```python
def ecdh_key_exchange(private_key, peer_public_key, curve):
    # Missing: Verify peer_public_key is on the curve
    shared_secret = private_key * peer_public_key
    return kdf(shared_secret.x)

# Attacker can inject points from a different curve with small subgroups
```

### Fixed Implementation
```python
def safe_ecdh_exchange(private_key, peer_public_key, curve):
    # 1. Validate public key
    if not is_point_on_curve(peer_public_key, curve):
        raise ValueError("Invalid public key")
    
    # 2. Check for small subgroup attacks
    if not is_point_at_infinity(peer_public_key * curve.cofactor):
        raise ValueError("Invalid public key (small subgroup)")
    
    # 3. Compute shared secret
    shared_secret = private_key * peer_public_key
    
    # 4. Key derivation with context
    return kdf(shared_secret.x, context="ECDH-v1")
```

## 11. Testing and Verification

### Differential Testing
```python
def test_ecdsa_sign_verify():
    private_key = gen_private_key_secure()
    public_key = private_key * G
    
    for _ in range(1000):
        message = os.urandom(32)
        signature = ecdsa_sign(message, private_key)
        assert ecdsa_verify(signature, message, public_key)
    
    # Test edge cases
    zero_sig = (0, 0)
    assert not ecdsa_verify(zero_sig, b"test", public_key)
    
    # Test invalid public keys
    invalid_pub = public_key + G  # Not the correct public key
    signature = ecdsa_sign(b"test", private_key)
    assert not ecdsa_verify(signature, b"test", invalid_pub)
```

### Property-Based Testing
```python
from hypothesis import given, strategies as st

@given(
    message=st.binary(min_size=1, max_size=1000),
    curve=st.sampled_from([SECP256R1, SECP384R1, SECP521R1])
)
def test_sign_verify_roundtrip(message, curve):

    public_key = private_key * curve.G
```

## 12. Secure Development Practices

### 1. Use High-Level Libraries
- **Recommended**: libsodium, OpenSSL 3.0+, BoringSSL
- **Avoid**: Implementing ECC from scratch

### 2. Enable Security Flags
```c
// Compiler flags for security
CFLAGS += -fstack-protector-strong -fPIE -fPIC -D_FORTIFY_SOURCE=2
LDFLAGS += -Wl,-z,now -Wl,-z,relro -pie

// Enable sanitizers for testing
TEST_CFLAGS = -fsanitize=address,undefined -fno-sanitize-recover
```

### 3. Continuous Fuzzing

## Conclusion

Elliptic curve cryptography offers strong security with smaller key sizes compared to RSA, but it comes with subtle implementation challenges. By understanding these common pitfalls and following secure coding practices, you can build more robust cryptographic systems.
