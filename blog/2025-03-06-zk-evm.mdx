    Opcode::ADD => {
        let a = stack.pop()?;
        let b = stack.pop()?;
        let sum = a.overflowing_add(b);
        stack.push(sum.0);
        // Constrain: a + b - sum = 0
        cs.enforce(
            || "add",
            |lc| lc + a.lc() + b.lc() - sum.lc(),
            |lc| lc + CS::one(),

## 3. State Management

### 3.1 Sparse Merkle Tree
```solidity
contract SparseMerkleTree {
    mapping(uint256 => bytes32) public nodes;
    
    function update(
        uint256 index,
        bytes32 value,
        bytes32[] calldata proof
    ) public {
        bytes32 node = keccak256(abi.encodePacked(index, value));
        for (uint256 i = 0; i < proof.length; i++) {
            if (index & (1 << i) == 0) {
                node = keccak256(abi.encodePacked(node, proof[i]));
            } else {
                node = keccak256(abi.encodePacked(proof[i], node));
            }
        }
        require(node == root, "Invalid proof");
    }
}
```
