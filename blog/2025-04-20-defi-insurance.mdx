# DeFi Insurance Landscape

## Insurance Models

### 1. Protocol-Owned Coverage

```solidity
// Simplified insurance pool
contract InsurancePool {
    IERC20 public immutable token;
    uint256 public totalCoverage;
    uint256 public premiumRate = 30; // 0.3%
    
    mapping(address => uint256) public coverage;
    mapping(address => uint256) public claims;
    
    function buyCoverage(uint256 amount, uint256 period) external {
        uint256 premium = (amount * premiumRate * period) / (365 days * 10000);
        token.transferFrom(msg.sender, address(this), premium);
        coverage[msg.sender] += amount;
        totalCoverage += amount;
    }
    
    function fileClaim(uint256 amount, bytes calldata proof) external {
        require(coverage[msg.sender] >= amount, "Insufficient coverage");
        require(verifyProof(proof), "Invalid claim");
        
        coverage[msg.sender] -= amount;
        claims[msg.sender] += amount;
        totalCoverage -= amount;
        
        token.transfer(msg.sender, amount);
    }
}
```
