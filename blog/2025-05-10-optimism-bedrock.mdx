# Optimism Bedrock Architecture

## Core Components

### 1. Rollup Node

```go
// Simplified rollup node
type RollupNode struct {
    ethClient  *ethclient.Client
    l1Head     *types.Header
    l2Head     *types.Header
    sequencer  *Sequencer
    verifier   *Verifier
    db         ethdb.Database
}

func (n *RollupNode) Start() {
    // Sync L1 blocks
    go n.syncL1()
    
    // Process L2 blocks
    go n.processL2()
}

func (n *RollupNode) syncL1() {
    // Subscribe to new L1 blocks
    headers := make(chan *types.Header)
    sub, _ := n.ethClient.SubscribeNewHead(context.Background(), headers)
    
    for {
        select {
        case header := <-headers:
            n.l1Head = header
            n.processL1Block(header)
        case err := <-sub.Err():
            log.Error("L1 subscription error", "err", err)
        }
    }
}
```

## Performance Optimizations

### 1. State Compression

```python
def compress_state(state: Dict[Address, Account]) -> bytes:
    # RLP encode state
    encoded = rlp.encode([
        (address, account.nonce, account.balance, account.storage_root, account.code_hash)
        for address, account in state.items()
    ])
    
    # Compress using zlib
    return zlib.compress(encoded, level=9)
```

### 2. Parallel Transaction Execution

```rust
async fn execute_block(block: Block) -> Result<(), Error> {
    let tasks: Vec<_> = block
        .transactions
        .into_iter()
        .map(|tx| tokio::spawn(execute_transaction(tx)))
        .collect();
    
    for task in tasks {
        task.await??;
    }
    
    Ok(())
}
```
